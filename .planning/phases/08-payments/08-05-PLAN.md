   export function SubscriptionStatus() {
     const [status, setStatus] = useState<UserAccess | null>(null);

     useEffect(() => {
       fetch('/api/subscription/status')
         .then(res => res.json())
         .then(setStatus);
     }, []);

     if (!status) return <Skeleton />;

     if (status.type === 'subscription') {
       return (
         <div>
           <p>Planas: {status.planName}</p>
           {!status.unlimited && (
             <p>Panaudota: {status.used}/{status.quota}</p>
           )}
           <p>Atnaujinimas: {format(status.periodEnd, 'yyyy-MM-dd')}</p>
           <Button onClick={handleManageSubscription}>
             Valdyti prenumeratą
           </Button>
         </div>
       );
     }
     // ... credits and free tier handling
   }
