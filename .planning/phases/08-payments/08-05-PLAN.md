Create API endpoint for fetching user's subscription status:

1. **Create app/api/subscription/status/route.ts:**
   ```typescript
   import { NextResponse } from 'next/server';
   import { createClient } from '@/lib/supabase/server';
   import { getUserAccess, getSubscriptionQuotaUsed } from '@/lib/stripe/subscriptions';
   import { getUserIdFromSupabase } from '@/lib/auth/helpers';

   export async function GET() {
     const supabase = await createClient();
     const { data: { user } } = await supabase.auth.getUser();

     if (!user) {
       return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
     }

     const userId = await getUserIdFromSupabase(user.id);
     if (!userId) {
       return NextResponse.json({ error: 'User not found' }, { status: 404 });
     }

     const access = await getUserAccess(userId);

     // Add quota usage for subscription users
     if (access.type === 'subscription' && !access.unlimited) {
       access.used = await getSubscriptionQuotaUsed(userId);
     }

     return NextResponse.json(access);
   }
   ```

2. **Response shape:**
   - Returns UserAccess type from lib/stripe/subscriptions.ts
   - Includes `used` count for quota-limited subscriptions
   - Used by SubscriptionStatus and other UI components

   ```tsx
   'use client';

   export function SubscriptionStatus() {
     const [status, setStatus] = useState<UserAccess | null>(null);
     const [loading, setLoading] = useState(false);

     useEffect(() => {
       fetch('/api/subscription/status')
         .then(res => res.json())
         .then(setStatus);
     }, []);

     async function handleManageSubscription() {
       setLoading(true);
       const res = await fetch('/api/stripe/portal', {
         method: 'POST',
       });
       const data = await res.json();
       if (data.url) {
         window.location.href = data.url;
       } else {
         toast.error('Klaida atidarant prenumeratos valdymą');
       }
       setLoading(false);
     }

     if (!status) return <Skeleton />;

     if (status.type === 'subscription') {
       return (
         <div>
           <p>Planas: {status.planName}</p>
           {!status.unlimited && (
             <p>Panaudota: {status.used}/{status.quota}</p>
           )}
           <p>Atnaujinimas: {format(status.periodEnd, 'yyyy-MM-dd')}</p>
           <Button onClick={handleManageSubscription} disabled={loading}>
             Valdyti prenumeratą
           </Button>
         </div>
       );
     }

     if (status.type === 'credits') {
       return (
         <div>
           <p>Kreditai: {status.credits}</p>
           <Link href="/pricing">Pirkti daugiau kreditų</Link>
         </div>
       );
     }

     // Free tier
     return (
       <div>
         <p>Planas: Nemokamas</p>
         <Link href="/pricing">Atnaujinti planą</Link>
       </div>
     );
   }
   ```
