  <action>
Create middleware.ts at project root with clerkMiddleware():

```typescript
// middleware.ts
import { clerkMiddleware, createRouteMatcher } from '@clerk/nextjs/server'

// Define protected routes (everything else is public by default)
// These will be used once dashboard/settings are built in later phases
const isProtectedRoute = createRouteMatcher([
  '/dashboard(.*)',
  '/settings(.*)',
  '/profile(.*)',
])

// Webhook and auth routes MUST be public
const isPublicRoute = createRouteMatcher([
  '/api/webhooks(.*)',
  '/sign-in(.*)',
  '/sign-up(.*)',
  '/',
  '/api/generate(.*)',
  '/api/generate-image(.*)',
  '/api/db-health(.*)',
])

export default clerkMiddleware(async (auth, req) => {
  // Always allow public routes
  if (isPublicRoute(req)) return

  // Protect designated routes
  if (isProtectedRoute(req)) {
    await auth.protect()
  }
})

export const config = {
  matcher: [
    // Skip Next.js internals and static files
    '/((?!_next|[^?]*\\.(?:html?|css|js(?!on)|jpe?g|webp|png|gif|svg|ttf|woff2?|ico|csv|docx?|xlsx?|zip|webmanifest)).*)',
    // Always run for API routes
    '/(api|trpc)(.*)',
  ],
}
```

CRITICAL NOTES:
- All routes are PUBLIC by default with clerkMiddleware()
- Only routes matching isProtectedRoute get protection via auth.protect()
- Webhook routes MUST be public (Clerk can't reach them otherwise)
- Current API routes (generate, generate-image, db-health) remain public for anonymous users
- Dashboard/settings routes are pre-configured for future phases

  children,
}: Readonly<{
  children: React.ReactNode;
}>) {
  return (
    <ClerkProvider>
      <html lang="lt">
        <body>
          {children}
          <Toaster
            position="bottom-center"

              style: {
                background: '#333',
                color: '#fff',
              },
            }}
          />

      </html>

  );
}
