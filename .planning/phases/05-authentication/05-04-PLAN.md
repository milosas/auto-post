```typescript
// app/components/AuthHeader.tsx
'use client'

import { useAuth, UserButton, SignInButton, SignUpButton } from '@clerk/nextjs'

export function AuthHeader() {
  const { isLoaded, isSignedIn } = useAuth()

  return (
    <header className="fixed top-0 left-0 right-0 z-50 bg-white/80 backdrop-blur-sm border-b border-gray-200">
      <div className="max-w-4xl mx-auto px-4 h-14 flex items-center justify-between">
        {/* Logo / Title */}
        <div className="font-semibold text-gray-900">
          Post Generator
        </div>

        {/* Auth Actions */}
        <div className="flex items-center gap-3">
          {!isLoaded ? (
            // Loading skeleton
            <div className="h-8 w-8 animate-pulse rounded-full bg-gray-200" />
          ) : isSignedIn ? (
            // Authenticated: Show UserButton with dropdown
            <UserButton
              afterSignOutUrl="/"
              appearance={{
                elements: {
                  avatarBox: 'h-8 w-8',
                  userButtonPopoverCard: 'shadow-lg',
                  userButtonPopoverActionButton: 'hover:bg-gray-100',
                },
              }}
            />
          ) : (
            // Not authenticated: Show Sign In / Sign Up buttons
            <>
              <SignInButton mode="modal">
                <button className="text-sm text-gray-600 hover:text-gray-900 transition-colors">
                  Prisijungti
                </button>
              </SignInButton>
              <SignUpButton mode="modal">
                <button className="text-sm bg-blue-600 text-white px-4 py-2 rounded-md hover:bg-blue-700 transition-colors">
                  Registruotis
                </button>
              </SignUpButton>
            </>
          )}
        </div>
      </div>
    </header>
  )
}
```

NOTES:
- Uses 'use client' directive - required for useAuth() hook
- useAuth() provides isLoaded (false during hydration) and isSignedIn
- UserButton provides: avatar, dropdown with profile link, sign out option
- SignInButton/SignUpButton with mode="modal" - opens Clerk modal without navigation
- afterSignOutUrl="/" - redirects to home after logout
- Lithuanian text: "Prisijungti" (Sign In), "Registruotis" (Sign Up)
- Styling matches app's visual language (blue primary, gray secondary)
- Fixed header with backdrop blur for modern feel
