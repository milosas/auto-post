```typescript
// app/api/webhooks/clerk/route.ts
import { verifyWebhook } from '@clerk/nextjs/server'
import { NextRequest } from 'next/server'
import { db } from '@/app/db'
import { users } from '@/app/db/schema'
import { eq } from 'drizzle-orm'

// Clerk webhook event types
type ClerkWebhookEvent = {
  type: string
  data: {
    id: string
    email_addresses: Array<{ email_address: string }>
    first_name: string | null
    last_name: string | null
    image_url: string | null
  }
}

export async function POST(req: NextRequest) {
  try {
    // Verify webhook signature using Svix (built into Clerk)
    // This validates svix-id, svix-timestamp, svix-signature headers
    const evt = await verifyWebhook(req) as ClerkWebhookEvent

    const { id, email_addresses, first_name, last_name, image_url } = evt.data
    const eventType = evt.type
    const email = email_addresses?.[0]?.email_address
    const name = [first_name, last_name].filter(Boolean).join(' ') || null

    console.log(`[Clerk Webhook] Event: ${eventType}, User: ${id}`)

    // Handle user.created - insert new user
    if (eventType === 'user.created') {
      if (!email) {
        console.error(`[Clerk Webhook] user.created missing email for ${id}`)
        return new Response('Missing email', { status: 400 })
      }

      // Check if user already exists (idempotency for webhook retries)
      const existingUser = await db.query.users.findFirst({
        where: eq(users.clerkId, id),
      })

      if (existingUser) {
        console.log(`[Clerk Webhook] User ${id} already exists, skipping insert`)
        return new Response('User already exists', { status: 200 })
      }

      await db.insert(users).values({
        clerkId: id,
        email,
        name,
        imageUrl: image_url,
      })

      console.log(`[Clerk Webhook] Created user: ${id}`)
      return new Response('User created', { status: 200 })
    }

    // Handle user.updated - update existing user
    if (eventType === 'user.updated') {
      if (!email) {
        console.error(`[Clerk Webhook] user.updated missing email for ${id}`)
        return new Response('Missing email', { status: 400 })
      }

      const result = await db.update(users)
        .set({
          email,
          name,
          imageUrl: image_url,
          updatedAt: new Date(),
        })
        .where(eq(users.clerkId, id))
        .returning({ id: users.id })

      if (result.length === 0) {
        // User doesn't exist yet - might be out-of-order webhook delivery
        // Create them instead (idempotent handling)
        await db.insert(users).values({
          clerkId: id,
          email,
          name,
          imageUrl: image_url,
        })
        console.log(`[Clerk Webhook] Created user on update event: ${id}`)
      } else {
        console.log(`[Clerk Webhook] Updated user: ${id}`)
      }

      return new Response('User updated', { status: 200 })
    }

    // Handle user.deleted - soft delete user
    if (eventType === 'user.deleted') {
      const result = await db.update(users)
        .set({ deletedAt: new Date() })
        .where(eq(users.clerkId, id))
        .returning({ id: users.id })

      if (result.length === 0) {
        console.log(`[Clerk Webhook] User ${id} not found for deletion`)
      } else {
        console.log(`[Clerk Webhook] Soft-deleted user: ${id}`)
      }

      // Return 200 even if user not found (idempotent)
      return new Response('User deleted', { status: 200 })
    }

    // Unknown event type - log and return 200 to stop retries
    console.log(`[Clerk Webhook] Unhandled event type: ${eventType}`)
    return new Response('Event type not handled', { status: 200 })

  } catch (err) {
    console.error('[Clerk Webhook] Error verifying webhook:', err)
    // Return 400 to trigger Clerk's exponential backoff retry
    return new Response('Error verifying webhook', { status: 400 })
  }
}
```

CRITICAL NOTES:
- verifyWebhook() validates Svix headers (svix-id, svix-timestamp, svix-signature)
- Always return 200 for successful processing (even if no-op) to stop retries
- Return 400 for verification failures to trigger retries
- Idempotency: Check for existing user before insert, handle out-of-order events
- Soft delete pattern matches Phase 4 schema design (deletedAt timestamp)
- This route MUST be public in middleware.ts (already configured in Plan 01)
